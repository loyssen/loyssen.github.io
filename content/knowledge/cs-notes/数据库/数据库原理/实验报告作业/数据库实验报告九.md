### 触发器
#### 一、创建和执行触发器
###### 1.1、交互式为数据表创建一个级联更新触发器TRIGGER_S

```
--====================================
--  Create database trigger template 
--====================================
USE jxsk奉恒斌
GO

IF EXISTS(
  SELECT *
    FROM sys.triggers
   WHERE name = 'TRRIGER_S'
     AND parent_class_desc = 'DATABASE'
)
	DROP TRIGGER TRRIGER_S ON DATABASE
GO

CREATE TRIGGER TRRIGER_S ON dbo.S 
	FOR update as if update(sno)
BEGIN
   declare @sno_new char(2), @sno_old char(2)
   select @sno_new=SNO from inserted
   select @sno_old=SNO from deleted
   update sc set SNO=@sno_new where SNO=@sno_old
END
GO
```

###### 1.2、交互式为数据库表SC创建一个限制更新触发器TRIGGER_SC
```
--====================================
--  Create database trigger template 
--====================================
USE jxsk奉恒斌
GO

IF EXISTS(
  SELECT *
    FROM sys.triggers
   WHERE name = 'TRRIGER_SC'
     AND parent_class_desc = 'dbo.SC'
)
	DROP TRIGGER TRRIGER_SC ON DATABASE
GO

CREATE TRIGGER TRRIGER_SC ON dbo.SC 
	FOR update as if update(sno)
BEGIN
   declare @sno_new char(2), @sno_old char(2),@scount int
   select @sno_new=SNO from inserted
   select @sno_old=SNO from deleted
   select @scount=count(*) from S where sno=@sno_old
   if @scount<>0
	rollback transaction
END
GO
```

###### 1.3、用T-SQL为数据库表SC创建一个触发器Score_SC_Tri
```
--====================================
--  Create database trigger template 
--====================================
USE jxsk奉恒斌
GO

IF EXISTS(
  SELECT *
    FROM sys.triggers
   WHERE name = 'Score_SC_Tri'
     AND parent_class_desc = 'DATABASE'
)
	DROP TRIGGER Score_SC_Tri ON DATABASE
GO


go
create trigger Score_SC_Tri ON dbo.SC
	for insert,update as
	declare @score tinyint
	select @score=SCORE from SC
	if @score>=0 and @score <=100
	begin
		print '操作成功'
	end
	else
	begin
		print '成绩不属于0~100！请重新操作。'
		rollback transaction
	end
go
```

###### 1.4、用T-SQL为数据库表C创建一个级联删除触发器TRGGER_DC
```
--====================================
--  Create database trigger template 
--====================================

  SELECT *
    FROM sys.triggers
   WHERE name = 'TRIGGER_DG'
     AND parent_class_desc = 'DATABASE'
)
	DROP TRIGGER TRIGGER_DG ON DATABASE
GO


go
create trigger TRIGGER_DG ON dbo.C
	for delete as
	declare @cno_del char(8)
	select @cno_del=CNO from deleted
	delete from sc where CNO=@cno_del
go
```
