###### 3.2、创建一个存储过程Pro_C，返回一个包含所有选修数据库课程的学生基本信息的游标。利用该存储过程，通过学生姓名查找学生王一山是否选修了数据库课程，如果选修了，则给出其成绩
```
use jxsk奉恒斌
go
create procedure Pro_C
@dcursor cursor varying output
as
begin
	set @dcursor=CURSOR for
		select SN,SCORE from S,SC,C
		where S.SNO=SC.SNO and SC.CNO=C.CNO
		and C.CN='数据库'
	open @dcursor
end
go

use jxsk奉恒斌

go
declare @dcursor_out cursor
declare @sn char(8), @score int
exec dbo.Pro_C @dcursor=@dcursor_out output
fetch next from @dcursor_out
	into @sn,@score
while @@fetch_status=0
begin 
	if(@sn='王一山')
	begin 
		print concat('王一山的数据库成绩：',@score)
		break
	end
	fetch next from @dcursor_out
		into @sn,@score
end
if(@@fetch_status!=0)
	print('王一山没有选修数据库')
close @dcursor_out
deallocate @dcursor_out
```
